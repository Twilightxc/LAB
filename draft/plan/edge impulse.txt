**2. โค้ด ESP32-S3 บน PlatformIO (`src/main.cpp`)**
ปรับปรุงจากโค้ดสัปดาห์ที่ 2 ให้รัน Inference ร่วมกับ Static Replay Buffer:

```cpp
#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>
#include <your_project_inferencing.h> // เปลี่ยนตามชื่อโปรเจกต์ Edge Impulse[cite: 3]

const char* ssid = "Wokwi-GUEST";
const char* password = "";
const char* mqtt_server = "demo.thingsboard.io";
const char* access_token = "YOUR_DEVICE_ACCESS_TOKEN";

WiFiClient espClient;
PubSubClient client(espClient);

// Static Buffer จำลองสัญญาณผิดปกติ (คัดลอกมาจากปุ่ม Copy Raw Features ใน Edge Impulse)[cite: 3, 5]
static const float test_vibration_features[] = {
    -0.12, -6.20, 7.90, -0.13, -6.19, 7.91, -0.14, -6.20, 7.92
    // ค่าจำลองจนครบขนาด Frame Size ของโมเดล[cite: 2]
};

// Callback สำหรับดึงข้อมูลเข้าสู่ Signal Wrapper ของ Edge Impulse
int raw_feature_get_data(size_t offset, size_t length, float *out_ptr) {
    memcpy(out_ptr, test_vibration_features + offset, length * sizeof(float));
    return 0;
}

void setup() {
    Serial.begin(115200);
    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
    Serial.println("\nWiFi Connected");
    client.setServer(mqtt_server, 1883);
}

void loop() {
    if (!client.connected()) {
        while (!client.connect("ESP32S3_AI_Node", access_token, NULL)) { delay(2000); }
    }
    client.loop();

    static unsigned long lastInference = 0;
    if (millis() - lastInference > 3000) {
        lastInference = millis();

        // 1. จัดเตรียม Signal Wrapper
        signal_t signal;
        signal.total_length = EI_CLASSIFIER_DSP_INPUT_FRAME_SIZE;
        signal.get_data = &raw_feature_get_data;

        // 2. รัน Inference ภายในตัวชิป ESP32-S3
        ei_impulse_result_t result = { 0 };
        EI_IMPULSE_ERROR r = run_classifier(&signal, &result, false);
        if (r != EI_IMPULSE_OK) {
            Serial.printf("ERR: Failed to run classifier (%d)\n", r);
            return;
        }

        // 3. วิเคราะห์ผลและแพ็ก JSON ส่ง ThingsBoard
        float anomaly_score = result.anomaly;
        const char* ai_status = (anomaly_score > 0.5) ? "FAULT" : "NORMAL";

        StaticJsonDocument<200> doc;
        doc["anomaly_score"] = anomaly_score;
        doc["ai_status"] = ai_status;
        doc["confidence"] = result.classification[0].value; // ความน่าจะเป็นของคลาสแรก

        char buffer[200];
        serializeJson(doc, buffer);
        client.publish("v1/devices/me/telemetry", buffer);
        Serial.printf("Sent Inference: %s\n", buffer);
    }
}

```

### ภาพรวมการทำงาน (Impulse Architecture)

```
[ Raw Sensor Data ] 
       │ (Windowing & Resampling)[cite: 1, 2]
       ▼
[ Processing Block (DSP) ] ──────► สกัด Features (ลดขนาดข้อมูล, O(n log n))[cite: 1, 3]
       │
       ▼
[ Learning Block (ML/NN) ] ──────► จำแนก Class + ตรวจจับ Anomaly[cite: 1, 2]
       │
       ▼
[ EON Compiler / Deployment ] ───► แปลงเป็น C++ / Arduino Library ลง ESP32-S3[cite: 1, 2, 3]

```

หากเครือข่ายหลุด อุปกรณ์จะไม่สามารถตรวจจับความผิดปกติได้

### โมดูลที่ 2: โครงสร้างของ Impulse (Input $\to$ DSP $\to$ Learning)

ในระบบของ Edge Impulse ไปป์ไลน์การประมวลผลจะเรียกว่า **Impulse** ซึ่งประกอบด้วย 3 บล็อกหลัก:

**1. Input Data Block (Windowing):**

* ทำการตัดแบ่งข้อมูล Time-series ออกเป็นหน้าต่างตามเวลา เช่น Window size 2000 ms และเลื่อนหน้าต่างทีละ Window increase 500 ms


* สามารถทำ Downsampling หรือ Upsampling ข้อมูลได้โดยอัตโนมัติหากสัญญาณบันทึกมาด้วยความถี่ที่ต่างกัน



**2. Processing Block (Digital Signal Processing - DSP):**

* มีหน้าที่สกัดคุณลักษณะ (Feature Extraction) ก่อนส่งให้โมเดล AI เพื่อลดภาระการคำนวณ ตัวอย่างเช่น การใช้ FFT เพื่อดึงความถี่มีความซับซ้อนระดับ $O(n \log n)$ 
ซึ่งประหยัดพลังงานกว่าการใช้ 1D Convolutional Neural Network ดิบ ๆ ที่ใช้ $O(n^2)$


* **Spectral Analysis (Spectral Features):** สกัดความถี่ (Frequency), กำลังของสัญญาณ (Power) และกรองสัญญาณรบกวน 
เหมาะสำหรับการเคลื่อนไหวหรือการสั่นสะเทือนของเครื่องจักร


* **Flatten:** สกัดค่าทางสถิติพื้นฐาน เช่น Average, Min, Max, RMS, Standard Deviation เหมาะสำหรับข้อมูลที่เปลี่ยนแปลงช้า (Slow-moving data) 
เช่น ก๊าซ อุณหภูมิ หรือความชื้น


* **Feature Explorer:** หน้าจอแสดงผลแบบ 3 มิติ เพื่อดูการกระจายตัวของคลัสเตอร์ข้อมูล ช่วยให้ประเมินได้ทันทีว่าสัญญาณแต่ละคลาสแยกออกจากกันได้ชัดเจนหรือไม่ก่อนเริ่มเทรน



**3. Learning Block:**

* การใช้ **Classification (Keras/NN)** เพื่อจำแนกคลาสที่ระบุไว้ในข้อมูลฝึกสอน (เช่น `idle`, `grind`, `pump`)


* การเพิ่ม **Anomaly Detection (K-Means Clustering):** สำหรับงานสั่นสะเทือน หากมีรูปแบบการสั่นใหม่ ๆ ที่ไม่เคยเทรนมาก่อน 
บล็อกนี้จะคำนวณระยะห่างเพื่อตีค่าออกมาเป็น `anomaly_score` สำหรับแจ้งเตือนความผิดปกติ



> **สคริปต์คำพูดผู้สอน:**
> "หัวใจสำคัญของ Edge Impulse คือการเอาบล็อก DSP มาประกบหน้า Machine Learning เราไม่โยนข้อมูลดิบทั้งก้อนเข้า Neural Network

### โมดูลที่ 3: การเพิ่มประสิทธิภาพด้วย EON Compiler & EON Tuner

**1. EON Compiler (กลไกการบีบอัดโมเดล):**

* โดยปกติการรันโมเดลบนไมโครคอนโทรลเลอร์ด้วย TensorFlow Lite for Microcontrollers (TFLM) จะต้องมี Interpreter คอยแปลคำสั่ง ซึ่งกิน RAM และ Flash ROM


* **EON Compiler** ทำการคอมไพล์โครงข่ายประสาทเทียมออกมาเป็น **ซอร์สโค้ด C++ โดยตรง** และตัด Interpreter ทิ้ง ทำให้ Linker ลบคำสั่งที่ไม่จำเป็นออกได้


* **ผลลัพธ์เชิงตัวเลข:** ช่วยประหยัด RAM ได้ 25–55% และลดขนาด Flash ROM ได้สูงสุดประมาณ 35% โดยยังคงค่าความแม่นยำเทียบเท่า TFLM


* *ข้อจำกัดทางเทคนิค:* โมเดลจะกลายเป็นส่วนหนึ่งของซอร์สโค้ดเฟิร์มแวร์ จึงปรับแก้โครงสร้างได้ยากขึ้นเมื่ออยู่ในหน้างาน (ต้องใช้วิธี OTA อัปเดตเฟิร์มแวร์ใหม่)



**2. EON Tuner (AutoML for Edge):**

* ทำหน้าที่ค้นหาคู่ชุดพารามิเตอร์ของ DSP และสถาปัตยกรรม Neural Network ที่เหมาะสมที่สุดกับข้อจำกัดของฮาร์ดแวร์


* ผู้ใช้กำหนดเงื่อนไข: Target Device (เช่น ESP32 / Cortex-M), ขีดจำกัดเวลาการคำนวณ (Latency) และเพดาน RAM/ROM


* ระบบจะทำการทดลองและพล็อตกราฟเปรียบเทียบระหว่างความแม่นยำ (Accuracy) กับปริมาณการใช้ทรัพยากร เพื่อให้ผู้พัฒนาเลือกจุดสมดุล (Trade-off) ที่ดีที่สุด

### โมดูลที่ 4: การนำไปใช้งานจริงบน PlatformIO และ Wokwi (Static Buffer Testing)

**แนวคิดการใช้งานจริง:**

* เมื่อเทรนและทดสอบบน Studio เรียบร้อย จะทำการ Export ออกมาเป็น **Arduino library (.zip)** หรือ **C++ library**

* โค้ดของ Edge Impulse มีโครงสร้างแบบ Standalone ไม่มี External Dependencies ซับซ้อน ทำให้คอมไพล์ผ่าน PlatformIO ได้ง่าย


* **เทคนิค Static Replay Buffer:** เนื่องจาก Wokwi Simulator ไม่สามารถเขย่าสร้างแรงสั่นสะเทือนเชิงกายภาพได้ เราจะใช้ความสามารถของฟังก์ชัน `run_classifier()` 
โดยดึงตัวอย่างสัญญาณดิบ (Raw Features) จาก Edge Impulse Studio มาจัดเก็บไว้ในหน่วยความจำของบอร์ด เพื่อทดสอบว่าฟังก์ชันคำนวณและประเมินผลได้สอดคล้องกับบน 
Studio หรือไม่ ก่อนส่งผลลัพธ์ผ่าน MQTT ไปยัง ThingsBoard
